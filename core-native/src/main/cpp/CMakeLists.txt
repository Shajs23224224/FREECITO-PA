cmake_minimum_required(VERSION 3.22.1)
project(omega LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
find_package(Threads REQUIRED)
add_library(omega_engine STATIC engine.cpp)
set_target_properties(omega_engine PROPERTIES POSITION_INDEPENDENT_CODE ON)
target_include_directories(omega_engine PUBLIC ${CMAKE_CURRENT_SOURCE_DIR})
target_link_libraries(omega_engine PUBLIC Threads::Threads)
target_compile_options(omega_engine PRIVATE -Wall -Wextra -Wpedantic)
if(NOT ANDROID)
  find_package(JNI REQUIRED)
endif()
add_library(omega_jni SHARED jni.cpp)
if(NOT ANDROID)
  target_include_directories(omega_jni PRIVATE ${JNI_INCLUDE_DIRS})
endif()
target_link_libraries(omega_jni PRIVATE omega_engine)
if(NOT ANDROID)
  enable_testing()
  add_executable(omega_native_tests ../../../../tests/native_test.cpp)
  target_link_libraries(omega_native_tests PRIVATE omega_engine)
  add_test(NAME native_engine COMMAND omega_native_tests)
endif()
